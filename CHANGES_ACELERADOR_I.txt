BUILD AND RUN (Jetson)
  cd ~/"Proyecto final"
  sed -i 's/\r$//' scripts/*.sh && chmod +x scripts/*.sh   # if edited on Windows
  rm -rf build && mkdir build && cd build && cmake .. && make -j$(nproc)
  ./test_cuda_correctness
  ./test_cuda_correctness ../data/frames/frame_000001.png ../data/frames/frame_000002.png
  ./bench_main ../data/frames/frame_000001.png ../data/frames/frame_000002.png --mode cuda
  for m in cuda cuda_opt cuda_shared; do ../scripts/profile_gpu.sh ./bench_main \
    ../data/frames/frame_000001.png ../data/frames/frame_000002.png $m 100; done
  for m in scalar neon cuda cuda_opt cuda_shared; do ../scripts/profile.sh ./bench_main \
    ../data/frames/frame_000001.png ../data/frames/frame_000002.png $m 100; done
